**Choix explicites (documentés dans les fiches de la tâche 3.1) :**
- Une seule recherche active par joueur (index partiel sur `player_id`) ; une seule commande de production active par planète, vaisseaux et défenses partageant le chantier (index partiel sur `planet_id`). Une file multi-commandes est un ajout ultérieur possible (M10), pas une exigence.
- Livraison **incrémentale** des unités : une commande de `quantity` unités de durée unitaire `unit_seconds` livre `min(quantity, floor((now − started_at) / unit_seconds))` unités à chaque règlement paresseux (même mécanisme que la production) ; l'événement `production_completed` à `started_at + quantity × unit_seconds` livre le reliquat et clôt la commande. Les unités livrées sont utilisables avant la fin du lot.
- Graviton : coût en **énergie disponible** (`Energy.Produced − Energy.Consumed ≥ 300000 × 3^(L−1)` sur la planète de lancement, satellites solaires inclus), aucun débit, durée 1 s.
- Réseau de recherche intergalactique : la durée utilise `lab_effective = lab(planète) + Σ des N meilleurs labos des autres planètes du joueur` avec `N = niveau IRN`, un labo distant ne comptant que si son niveau ≥ prérequis labo de la recherche ; `LaboratoryBonus` (ruleset) multiplie la contribution des labos : `1 + lab_effective × LaboratoryBonus`.
- Prérequis génériques par un paquet feuille `internal/domain/prereq` (kind + id + niveau), évitant les cycles d'import entre `building`, `research` et `unit`. Les bâtiments gagnent leurs prérequis de recherche classiques (nanites ← ordinateur 10, terraformeur ← énergie 12).
- Satellites solaires : unités qui produisent `floor((max_temperature + 160) / 6)` énergie chacun (borné à [0, 50]), intégrés à `economy.CalculateRates` via `Levels.SolarSatellites`. L'énergie des satellites livrés pendant un intervalle compte à partir du règlement suivant (approximation bornée, documentée).
- Exclusivité des installations (fidélité classique) : une recherche est refusée tant que `research_lab` est dans la file de construction de la planète de lancement (`ErrLaboratoryBusy`) et, symétriquement, `StartConstruction(research_lab)` est refusé pendant une recherche active du joueur ; `shipyard` et `nanite_factory` ne se construisent pas pendant une commande de production active et une commande est refusée pendant leur construction (`ErrShipyardBusy`). L'erreur économie commune est `appeconomy.ErrFacilityBusy`.
- Plafond dur `unit.MaximumOrderQuantity = 1_000_000` unités par commande (documenté), en plus des vérifications de débordement.
- `production_orders` porte la contrainte `CHECK (state <> 'completed' OR delivered = quantity)` : une commande close a tout livré.

```go
package prereq

import ("errors"; "fmt")

type Kind string

const (
	Building Kind = "building"
	Research Kind = "research"
)

type Requirement struct {
	Kind  Kind
	ID    string
	Level int
}

type LevelFunc func(kind Kind, id string) int

var (
	ErrUnmet       = errors.New("prereq: prerequisites are not met")
	ErrCycle       = errors.New("prereq: requirement graph contains a cycle")
	ErrUnknownNode = errors.New("prereq: requirement references an unknown node")
)

func Check(requirements []Requirement, levelOf LevelFunc) error {
	for _, requirement := range requirements {
		if levelOf(requirement.Kind, requirement.ID) < requirement.Level {
			return fmt.Errorf("%w: %s %s level %d", ErrUnmet, requirement.Kind, requirement.ID, requirement.Level)
		}
	}
	return nil
}

// ValidateGraph runs Kahn's algorithm over node -> dependencies.
func ValidateGraph(edges map[string][]string) error {
	indegree := make(map[string]int, len(edges))
	for node, dependencies := range edges {
		if _, ok := indegree[node]; !ok {
			indegree[node] = 0
		}
		for _, dependency := range dependencies {
			if _, known := edges[dependency]; !known {
				return fmt.Errorf("%w: %s -> %s", ErrUnknownNode, node, dependency)
			}
			indegree[node]++
		}
	}
	queue := make([]string, 0, len(edges))
	for node, degree := range indegree {
		if degree == 0 {
			queue = append(queue, node)
		}
	}
	visited := 0
	for len(queue) > 0 {
		current := queue[0]
		queue = queue[1:]
		visited++
		for node, dependencies := range edges {
			for _, dependency := range dependencies {
				if dependency == current {
					indegree[node]--
					if indegree[node] == 0 {
						queue = append(queue, node)
					}
				}
			}
		}
	}
	if visited != len(edges) {
		return ErrCycle
	}
	return nil
}
```

```go
func (c Catalogue) Duration(cost economy.Resources, laboratories Laboratories, minimumLab int, bonus, speed float64) (time.Duration, error) {
	if err := cost.Validate(); err != nil || laboratories.Local < 0 || laboratories.NetworkLevel < 0 || bonus <= 0 || speed <= 0 || math.IsNaN(bonus*speed) || math.IsInf(bonus*speed, 0) {
		return 0, errors.New("research: invalid duration input")
	}
	eligible := make([]int, 0, len(laboratories.Remote))
	for _, level := range laboratories.Remote {
		if level >= minimumLab && level > 0 {
			eligible = append(eligible, level)
		}
	}
	sort.Sort(sort.Reverse(sort.IntSlice(eligible)))
	effective := laboratories.Local
	for index := 0; index < laboratories.NetworkLevel && index < len(eligible); index++ {
		effective += eligible[index]
	}
	if cost.Metal > math.MaxInt64-cost.Crystal {
		return 0, errors.New("research: work overflow")
	}
	seconds := math.Floor(3600 * float64(cost.Metal+cost.Crystal) / (1000 * (1 + float64(effective)*bonus) * speed))
	if math.IsNaN(seconds) || math.IsInf(seconds, 0) || seconds > float64(math.MaxInt64/int64(time.Second)) {
		return 0, errors.New("research: duration overflow")
	}
	if seconds < 1 {
		seconds = 1
	}
	return time.Duration(int64(seconds)) * time.Second, nil
}
```

```go
func Deliverable(quantity, delivered, unitSeconds int64, startedAt, now time.Time) (int64, error) {
	if quantity <= 0 || delivered < 0 || delivered > quantity || unitSeconds <= 0 {
		return 0, ErrInvalidQuantity
	}
	if now.Before(startedAt) {
		return 0, errors.New("unit: delivery time precedes order start")
	}
	elapsed := int64(now.Sub(startedAt) / time.Second)
	produced := elapsed / unitSeconds
	if produced > quantity {
		produced = quantity
	}
	if produced <= delivered {
		return 0, nil
	}
	return produced - delivered, nil
}
```

```go
func TestProductionOrderDeliversIncrementallyAndCompletesOnce(t *testing.T) {
	ctx := context.Background()
	clock := appclock.NewFake(time.Date(2042, time.September, 10, 11, 12, 13, 0, time.UTC))
	database := economyDatabase(t, ctx, 1)
	world := newWorld(t, database, clock)
	principal := appauth.Principal{AccountID: 1}
	planet, _ := world.Economy.CreateEmpire(ctx, principal, "Captain")
	setBuilding(t, database, planet.ID, "shipyard", 1)
	setResearch(t, database, 1, "combustion_drive", 1)
	setResources(t, database, planet.ID, 9000, 3000, 0)

	order, err := world.Shipyard.Order(ctx, principal, planet.ID, unit.LightFighter, 3, "order-1")
	if err != nil || order.Quantity != 3 || order.UnitDuration != 2880*time.Second || order.TotalCost != (economy.Resources{Metal: 9000, Crystal: 3000}) {
		t.Fatalf("Order() = %+v %v", order, err)
	}
	assertSingleValue(t, database, "SELECT metal FROM planet_resources WHERE planet_id = 1", 0)
	if _, err := world.Shipyard.Order(ctx, principal, planet.ID, unit.RocketLauncher, 1, "order-2"); !errors.Is(err, appshipyard.ErrShipyardBusy) {
		t.Fatalf("second order error = %v", err)
	}
	clock.Advance(2 * 2880 * time.Second)
	overview, err := world.Shipyard.Overview(ctx, principal, planet.ID, unit.Ship)
	if err != nil || overview.Inventory[unit.LightFighter] != 2 || overview.Active == nil || overview.Active.Delivered != 2 {
		t.Fatalf("partial delivery = %+v %v", overview, err)
	}
	clock.Advance(2880 * time.Second)
	if _, err := world.Processor.CompleteDue(ctx, 10); err != nil {
		t.Fatal(err)
	}
	assertSingleValue(t, database, "SELECT quantity FROM planet_units WHERE planet_id = 1 AND unit_id = 'light_fighter'", 3)
	assertSingleValue(t, database, "SELECT COUNT(*) FROM production_orders WHERE state = 'completed'", 1)
	if _, err := database.Write().ExecContext(ctx, "UPDATE scheduled_events SET state = 'pending', processed_at = NULL WHERE event_type = 'production_completed'"); err != nil {
		t.Fatal(err)
	}
	if _, err := world.Processor.CompleteDue(ctx, 10); err != nil {
		t.Fatal(err)
	}
	assertSingleValue(t, database, "SELECT quantity FROM planet_units WHERE planet_id = 1 AND unit_id = 'light_fighter'", 3)
}
```
